
DO $$
DECLARE
    r RECORD;
BEGIN
    -- Hapus seluruh tabel pada schema public
    FOR r IN
        SELECT tablename
        FROM pg_tables
        WHERE schemaname = 'public'
    LOOP
        EXECUTE
            'DROP TABLE IF EXISTS public.'
            || quote_ident(r.tablename)
            || ' CASCADE';
    END LOOP;
END $$;
